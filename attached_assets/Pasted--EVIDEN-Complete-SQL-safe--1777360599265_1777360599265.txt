-- ── 4. Security-definer function (fixes infinite recursion) ──
CREATE OR REPLACE FUNCTION public.is_general_admin()
RETURNS boolean
LANGUAGE sql
SECURITY DEFINER
SET search_path = public
AS $$
  SELECT EXISTS (
    SELECT 1 FROM public.user_roles
    WHERE LOWER(email) = LOWER(auth.jwt() ->> 'email')
      AND role = 'general_admin'
  );
$$;

-- ── 5. Policies for user_roles ────────────────────────────────
-- Any logged-in user can read (so the app can detect their role on login)
CREATE POLICY "roles_read" ON user_roles
  FOR SELECT USING (auth.role() = 'authenticated');

-- Only general_admin can insert / update / delete
CREATE POLICY "roles_write" ON user_roles
  FOR ALL USING (public.is_general_admin());

-- ── 6. Policies for admin_logs ────────────────────────────────
-- Any logged-in user can write logs
CREATE POLICY "logs_insert" ON admin_logs
  FOR INSERT WITH CHECK (auth.role() = 'authenticated');

-- Only general_admin can read logs
CREATE POLICY "logs_read" ON admin_logs
  FOR SELECT USING (public.is_general_admin());
